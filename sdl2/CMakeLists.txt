cmake_minimum_required(VERSION 3.16)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/TotalCrossWindowsStaticRuntime.cmake")

project(totalcross-sdl2-static
  VERSION 2.32.8
  LANGUAGES C
)

include(ExternalProject)
include(GNUInstallDirs)

set(TC_SDL2_SOURCE_DIR "" CACHE PATH "Directory containing SDL 2.32.8 sources")
set(TC_SDL2_SOURCE_REVISION
  "98d1f3a45aae568ccd6ed5fec179330f47d4d356"
  CACHE STRING "Immutable SDL 2.32.8 source revision")
set(TC_SDL2_SOURCE_URL
  "https://codeload.github.com/libsdl-org/SDL/tar.gz/${TC_SDL2_SOURCE_REVISION}"
  CACHE STRING "SDL 2.32.8 source archive URL")
set(TC_SDL2_SOURCE_SHA256
  "be55f7015a5599faa869b20078de83df7e83c35585ee1c3ee203a0e58eefebbd"
  CACHE STRING "Expected SHA-256 for the SDL 2.32.8 source archive")

set(TC_SDL2_EXTERNAL_INSTALL_DIR "${CMAKE_BINARY_DIR}/sdl2-install")
set(TC_SDL2_CMAKE_ARGS
  "-DCMAKE_INSTALL_PREFIX=<INSTALL_DIR>"
  "-DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}"
  "-DCMAKE_POSITION_INDEPENDENT_CODE=ON"
  "-DCMAKE_POLICY_DEFAULT_CMP0091=NEW"
  "-DSDL_SHARED=OFF"
  "-DSDL_STATIC=ON"
  "-DSDL_STATIC_PIC=ON"
  "-DSDL_TEST=OFF"
  "-DSDL_TESTS=OFF"
  "-DSDL_INSTALL_TESTS=OFF"
  "-DSDL2_DISABLE_SDL2MAIN=ON"
  "-DSDL_FORCE_STATIC_VCRT=ON"
  "-DSDL_INSTALL_CMAKEDIR=${CMAKE_INSTALL_LIBDIR}/cmake/SDL2"
)
set(TC_SDL2_EXTERNAL_PROJECT_OPTIONS)

if(DEFINED CMAKE_MSVC_RUNTIME_LIBRARY AND NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "")
  list(APPEND TC_SDL2_CMAKE_ARGS
    "-DCMAKE_MSVC_RUNTIME_LIBRARY=${CMAKE_MSVC_RUNTIME_LIBRARY}")
endif()

if(CMAKE_GENERATOR_PLATFORM OR CMAKE_GENERATOR_TOOLSET)
  list(APPEND TC_SDL2_EXTERNAL_PROJECT_OPTIONS
    CMAKE_GENERATOR "${CMAKE_GENERATOR}"
  )
endif()
if(CMAKE_GENERATOR_PLATFORM)
  list(APPEND TC_SDL2_EXTERNAL_PROJECT_OPTIONS
    CMAKE_GENERATOR_PLATFORM "${CMAKE_GENERATOR_PLATFORM}"
  )
endif()
if(CMAKE_GENERATOR_TOOLSET)
  list(APPEND TC_SDL2_EXTERNAL_PROJECT_OPTIONS
    CMAKE_GENERATOR_TOOLSET "${CMAKE_GENERATOR_TOOLSET}"
  )
endif()

if(DEFINED CMAKE_TOOLCHAIN_FILE)
  list(APPEND TC_SDL2_CMAKE_ARGS "-DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE}")
endif()
if(CMAKE_OSX_SYSROOT)
  list(APPEND TC_SDL2_CMAKE_ARGS "-DCMAKE_OSX_SYSROOT=${CMAKE_OSX_SYSROOT}")
endif()
if(CMAKE_OSX_ARCHITECTURES)
  list(APPEND TC_SDL2_CMAKE_ARGS "-DCMAKE_OSX_ARCHITECTURES=${CMAKE_OSX_ARCHITECTURES}")
endif()
if(CMAKE_OSX_DEPLOYMENT_TARGET)
  list(APPEND TC_SDL2_CMAKE_ARGS "-DCMAKE_OSX_DEPLOYMENT_TARGET=${CMAKE_OSX_DEPLOYMENT_TARGET}")
endif()
if(CMAKE_SYSTEM_PROCESSOR)
  list(APPEND TC_SDL2_CMAKE_ARGS "-DCMAKE_SYSTEM_PROCESSOR=${CMAKE_SYSTEM_PROCESSOR}")
elseif(CMAKE_OSX_ARCHITECTURES)
  list(GET CMAKE_OSX_ARCHITECTURES 0 TC_SDL2_SYSTEM_PROCESSOR)
  list(APPEND TC_SDL2_CMAKE_ARGS "-DCMAKE_SYSTEM_PROCESSOR=${TC_SDL2_SYSTEM_PROCESSOR}")
endif()
if(CMAKE_SYSTEM_NAME)
  list(APPEND TC_SDL2_CMAKE_ARGS "-DCMAKE_SYSTEM_NAME=${CMAKE_SYSTEM_NAME}")
endif()

if(TC_SDL2_SOURCE_DIR)
  get_filename_component(TC_SDL2_SOURCE_DIR_ABSOLUTE "${TC_SDL2_SOURCE_DIR}" ABSOLUTE)
  if(NOT EXISTS "${TC_SDL2_SOURCE_DIR_ABSOLUTE}/include/SDL.h" OR
     NOT EXISTS "${TC_SDL2_SOURCE_DIR_ABSOLUTE}/LICENSE.txt")
    message(FATAL_ERROR
      "TC_SDL2_SOURCE_DIR must contain include/SDL.h and LICENSE.txt")
  endif()
  ExternalProject_Add(sdl2_external
    ${TC_SDL2_EXTERNAL_PROJECT_OPTIONS}
    SOURCE_DIR "${TC_SDL2_SOURCE_DIR_ABSOLUTE}"
    BINARY_DIR "${CMAKE_BINARY_DIR}/sdl2-build"
    INSTALL_DIR "${TC_SDL2_EXTERNAL_INSTALL_DIR}"
    CMAKE_ARGS ${TC_SDL2_CMAKE_ARGS}
  )
else()
  ExternalProject_Add(sdl2_external
    ${TC_SDL2_EXTERNAL_PROJECT_OPTIONS}
    URL "${TC_SDL2_SOURCE_URL}"
    URL_HASH "SHA256=${TC_SDL2_SOURCE_SHA256}"
    DOWNLOAD_EXTRACT_TIMESTAMP FALSE
    BINARY_DIR "${CMAKE_BINARY_DIR}/sdl2-build"
    INSTALL_DIR "${TC_SDL2_EXTERNAL_INSTALL_DIR}"
    CMAKE_ARGS ${TC_SDL2_CMAKE_ARGS}
  )
endif()

install(DIRECTORY "${TC_SDL2_EXTERNAL_INSTALL_DIR}/include/"
  DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
)
install(DIRECTORY "${TC_SDL2_EXTERNAL_INSTALL_DIR}/lib/"
  DESTINATION "${CMAKE_INSTALL_LIBDIR}"
)
install(FILES "${TC_SDL2_EXTERNAL_INSTALL_DIR}/share/licenses/SDL2/LICENSE.txt"
  DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/sdl2")
