cmake_minimum_required(VERSION 3.16)

project(totalcross-libjpeg-turbo-static
  VERSION 3.1.4.1
  LANGUAGES C
)

include(ExternalProject)
include(GNUInstallDirs)

set(LIBJPEG_TURBO_SOURCE_DIR "" CACHE PATH "Directory containing libjpeg-turbo sources")
set(LIBJPEG_TURBO_GIT_REPOSITORY "https://github.com/libjpeg-turbo/libjpeg-turbo.git" CACHE STRING "libjpeg-turbo Git repository")
set(LIBJPEG_TURBO_GIT_TAG "3.1.4.1" CACHE STRING "libjpeg-turbo Git tag")

set(TC_LIBJPEG_TURBO_EXTERNAL_INSTALL_DIR "${CMAKE_BINARY_DIR}/libjpeg-turbo-install")
set(TC_LIBJPEG_TURBO_PATCH_SCRIPT "${CMAKE_BINARY_DIR}/patch-libjpeg-turbo.cmake")

file(WRITE "${TC_LIBJPEG_TURBO_PATCH_SCRIPT}" [==[
if(NOT DEFINED TC_LIBJPEG_TURBO_SOURCE_DIR)
  message(FATAL_ERROR "TC_LIBJPEG_TURBO_SOURCE_DIR is required")
endif()

set(TC_LIBJPEG_TURBO_CMAKELISTS "${TC_LIBJPEG_TURBO_SOURCE_DIR}/CMakeLists.txt")
if(EXISTS "${TC_LIBJPEG_TURBO_CMAKELISTS}")
  file(READ "${TC_LIBJPEG_TURBO_CMAKELISTS}" TC_LIBJPEG_TURBO_CMAKELISTS_CONTENTS)
  string(REPLACE
    [=[string(TOLOWER ${CMAKE_SYSTEM_PROCESSOR} CMAKE_SYSTEM_PROCESSOR_LC)]=]
    [=[string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" CMAKE_SYSTEM_PROCESSOR_LC)]=]
    TC_LIBJPEG_TURBO_CMAKELISTS_PATCHED
    "${TC_LIBJPEG_TURBO_CMAKELISTS_CONTENTS}"
  )
  if(NOT TC_LIBJPEG_TURBO_CMAKELISTS_PATCHED STREQUAL TC_LIBJPEG_TURBO_CMAKELISTS_CONTENTS)
    file(WRITE "${TC_LIBJPEG_TURBO_CMAKELISTS}" "${TC_LIBJPEG_TURBO_CMAKELISTS_PATCHED}")
  endif()
endif()
]==])

set(TC_LIBJPEG_TURBO_CMAKE_ARGS
  "-DCMAKE_INSTALL_PREFIX=<INSTALL_DIR>"
  "-DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}"
  "-DCMAKE_POSITION_INDEPENDENT_CODE=ON"
  "-DENABLE_SHARED=OFF"
  "-DENABLE_STATIC=ON"
  "-DWITH_TOOLS=OFF"
  "-DWITH_TESTS=OFF"
  "-DWITH_JAVA=OFF"
  "-DWITH_TURBOJPEG=ON"
)
set(TC_LIBJPEG_TURBO_EXTERNAL_PROJECT_OPTIONS)

if(CMAKE_GENERATOR_PLATFORM)
  list(APPEND TC_LIBJPEG_TURBO_EXTERNAL_PROJECT_OPTIONS
    CMAKE_GENERATOR_PLATFORM "${CMAKE_GENERATOR_PLATFORM}"
  )
endif()
if(CMAKE_GENERATOR_TOOLSET)
  list(APPEND TC_LIBJPEG_TURBO_EXTERNAL_PROJECT_OPTIONS
    CMAKE_GENERATOR_TOOLSET "${CMAKE_GENERATOR_TOOLSET}"
  )
endif()

if(DEFINED CMAKE_TOOLCHAIN_FILE)
  list(APPEND TC_LIBJPEG_TURBO_CMAKE_ARGS "-DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE}")
endif()
if(DEFINED ANDROID_ABI)
  list(APPEND TC_LIBJPEG_TURBO_CMAKE_ARGS "-DANDROID_ABI=${ANDROID_ABI}")
endif()
if(DEFINED ANDROID_PLATFORM)
  list(APPEND TC_LIBJPEG_TURBO_CMAKE_ARGS "-DANDROID_PLATFORM=${ANDROID_PLATFORM}")
endif()
if(DEFINED ANDROID_USE_LEGACY_TOOLCHAIN_FILE)
  list(APPEND TC_LIBJPEG_TURBO_CMAKE_ARGS "-DANDROID_USE_LEGACY_TOOLCHAIN_FILE=${ANDROID_USE_LEGACY_TOOLCHAIN_FILE}")
endif()
if(CMAKE_OSX_SYSROOT)
  list(APPEND TC_LIBJPEG_TURBO_CMAKE_ARGS "-DCMAKE_OSX_SYSROOT=${CMAKE_OSX_SYSROOT}")
endif()
if(CMAKE_OSX_ARCHITECTURES)
  list(APPEND TC_LIBJPEG_TURBO_CMAKE_ARGS "-DCMAKE_OSX_ARCHITECTURES=${CMAKE_OSX_ARCHITECTURES}")
endif()
if(CMAKE_OSX_DEPLOYMENT_TARGET)
  list(APPEND TC_LIBJPEG_TURBO_CMAKE_ARGS "-DCMAKE_OSX_DEPLOYMENT_TARGET=${CMAKE_OSX_DEPLOYMENT_TARGET}")
endif()
if(DEFINED CMAKE_XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED)
  list(APPEND TC_LIBJPEG_TURBO_CMAKE_ARGS "-DCMAKE_XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED=${CMAKE_XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED}")
endif()
if(CMAKE_SYSTEM_PROCESSOR)
  list(APPEND TC_LIBJPEG_TURBO_CMAKE_ARGS "-DCMAKE_SYSTEM_PROCESSOR=${CMAKE_SYSTEM_PROCESSOR}")
elseif(CMAKE_OSX_ARCHITECTURES)
  list(GET CMAKE_OSX_ARCHITECTURES 0 TC_LIBJPEG_TURBO_SYSTEM_PROCESSOR)
  list(APPEND TC_LIBJPEG_TURBO_CMAKE_ARGS "-DCMAKE_SYSTEM_PROCESSOR=${TC_LIBJPEG_TURBO_SYSTEM_PROCESSOR}")
endif()
if(CMAKE_SYSTEM_NAME)
  list(APPEND TC_LIBJPEG_TURBO_CMAKE_ARGS "-DCMAKE_SYSTEM_NAME=${CMAKE_SYSTEM_NAME}")
endif()

if(LIBJPEG_TURBO_SOURCE_DIR)
  ExternalProject_Add(libjpeg_turbo_external
    ${TC_LIBJPEG_TURBO_EXTERNAL_PROJECT_OPTIONS}
    SOURCE_DIR "${LIBJPEG_TURBO_SOURCE_DIR}"
    BINARY_DIR "${CMAKE_BINARY_DIR}/libjpeg-turbo-build"
    INSTALL_DIR "${TC_LIBJPEG_TURBO_EXTERNAL_INSTALL_DIR}"
    PATCH_COMMAND "${CMAKE_COMMAND}" "-DTC_LIBJPEG_TURBO_SOURCE_DIR=<SOURCE_DIR>" -P "${TC_LIBJPEG_TURBO_PATCH_SCRIPT}"
    CMAKE_ARGS ${TC_LIBJPEG_TURBO_CMAKE_ARGS}
  )
else()
  ExternalProject_Add(libjpeg_turbo_external
    ${TC_LIBJPEG_TURBO_EXTERNAL_PROJECT_OPTIONS}
    GIT_REPOSITORY "${LIBJPEG_TURBO_GIT_REPOSITORY}"
    GIT_TAG "${LIBJPEG_TURBO_GIT_TAG}"
    GIT_SHALLOW TRUE
    GIT_PROGRESS TRUE
    USES_TERMINAL_DOWNLOAD TRUE
    BINARY_DIR "${CMAKE_BINARY_DIR}/libjpeg-turbo-build"
    INSTALL_DIR "${TC_LIBJPEG_TURBO_EXTERNAL_INSTALL_DIR}"
    PATCH_COMMAND "${CMAKE_COMMAND}" "-DTC_LIBJPEG_TURBO_SOURCE_DIR=<SOURCE_DIR>" -P "${TC_LIBJPEG_TURBO_PATCH_SCRIPT}"
    CMAKE_ARGS ${TC_LIBJPEG_TURBO_CMAKE_ARGS}
  )
endif()

install(DIRECTORY "${TC_LIBJPEG_TURBO_EXTERNAL_INSTALL_DIR}/include/"
  DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
  FILES_MATCHING PATTERN "*.h"
)

install(DIRECTORY "${TC_LIBJPEG_TURBO_EXTERNAL_INSTALL_DIR}/lib/"
  DESTINATION "${CMAKE_INSTALL_LIBDIR}"
  FILES_MATCHING
    PATTERN "*.a"
    PATTERN "*.lib"
)
