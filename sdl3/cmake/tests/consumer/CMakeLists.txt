cmake_minimum_required(VERSION 3.16)
project(sdl3_depot_consumer C)

if(NOT DEFINED DEPOT_TOOLS_ROOT)
  message(FATAL_ERROR "DEPOT_TOOLS_ROOT is required")
endif()
if(SDL3_TEST_CONFIG_ONLY)
  if(NOT SDL3_DIR)
    message(FATAL_ERROR "SDL3_DIR is required for CONFIG-only validation")
  endif()
  find_package(SDL3 CONFIG REQUIRED)
else()
  list(PREPEND CMAKE_MODULE_PATH "${DEPOT_TOOLS_ROOT}/sdl3/cmake")
  if(SDL3_TEST_AUTO_FETCH)
    include(AutoFetchSDL3)
    tcvm_auto_fetch_sdl3()
  endif()
  find_package(SDL3 REQUIRED)
endif()

if(NOT TARGET SDL3::SDL3 OR NOT TARGET SDL3::SDL3-static)
  message(FATAL_ERROR "Expected SDL3::SDL3 and SDL3::SDL3-static")
endif()
if(NOT SDL3_TEST_CONFIG_ONLY)
  get_target_property(SDL3_TEST_RESOLVED_ROOT SDL3::SDL3-static TC_SDL3_DEPOT_ROOT)
  if(NOT SDL3_TEST_RESOLVED_ROOT STREQUAL SDL3_DEPOT_ROOT)
    message(FATAL_ERROR
      "SDL3 resolved outside the requested depot root: ${SDL3_TEST_RESOLVED_ROOT}")
  endif()
endif()

add_executable(sdl3-depot-consumer main.c)
if(SDL3_TEST_EXPLICIT_STATIC)
  target_link_libraries(sdl3-depot-consumer PRIVATE SDL3::SDL3-static)
else()
  target_link_libraries(sdl3-depot-consumer PRIVATE SDL3::SDL3)
endif()
