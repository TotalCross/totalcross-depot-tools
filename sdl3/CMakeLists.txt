cmake_minimum_required(VERSION 3.16)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/TotalCrossWindowsStaticRuntime.cmake")

project(totalcross-sdl3-static
  VERSION 3.4.16
  LANGUAGES C
)

include(ExternalProject)
include(GNUInstallDirs)

set(TC_SDL3_SOURCE_DIR "" CACHE PATH "Directory containing SDL 3.4.16 sources")
set(TC_SDL3_SOURCE_REVISION
  "fa2c02bb6e21974a89ea9824bc53c9932abe5f9c"
  CACHE STRING "Immutable SDL 3.4.16 source revision")
set(TC_SDL3_SOURCE_URL
  "https://codeload.github.com/libsdl-org/SDL/tar.gz/${TC_SDL3_SOURCE_REVISION}"
  CACHE STRING "SDL 3.4.16 source archive URL")
set(TC_SDL3_SOURCE_SHA256
  "d246967187a4f37cc850608dbaf5cff9d1df4d731259903d25539755cdfcc657"
  CACHE STRING "Expected SHA-256 for the SDL 3.4.16 source archive")

set(TC_SDL3_EXTERNAL_INSTALL_DIR "${CMAKE_BINARY_DIR}/sdl3-install")
set(TC_SDL3_CMAKE_ARGS
  "-DCMAKE_INSTALL_PREFIX=<INSTALL_DIR>"
  "-DCMAKE_BUILD_TYPE=${CMAKE_BUILD_TYPE}"
  "-DCMAKE_POSITION_INDEPENDENT_CODE=ON"
  "-DCMAKE_POLICY_DEFAULT_CMP0091=NEW"
  "-DBUILD_SHARED_LIBS=OFF"
  "-DSDL_SHARED=OFF"
  "-DSDL_STATIC=ON"
  "-DSDL_INSTALL=ON"
  "-DSDL_INSTALL_CPACK=OFF"
  "-DSDL_INSTALL_DOCS=OFF"
  "-DSDL_RELOCATABLE=ON"
  "-DSDL_TEST_LIBRARY=OFF"
  "-DSDL_TESTS=OFF"
  "-DSDL_INSTALL_TESTS=OFF"
  "-DSDL_EXAMPLES=OFF"
  "-DSDL_LIBC=ON"
  "-DSDL_VIDEO=ON"
  "-DSDL_OPENGL=ON"
  "-DSDL_OPENGLES=ON"
  "-DSDL_VULKAN=ON"
  "-DSDL_DUMMYVIDEO=ON"
  "-DSDL_OFFSCREEN=ON"
  "-DSDL_AUDIO=OFF"
  "-DSDL_GPU=OFF"
  "-DSDL_RENDER=OFF"
  "-DSDL_CAMERA=OFF"
  "-DSDL_JOYSTICK=OFF"
  "-DSDL_HAPTIC=OFF"
  "-DSDL_HIDAPI=OFF"
  "-DSDL_POWER=OFF"
  "-DSDL_SENSOR=OFF"
  "-DSDL_DIALOG=OFF"
  "-DSDL_TRAY=OFF"
  "-DSDL_INSTALL_CMAKEDIR=${CMAKE_INSTALL_LIBDIR}/cmake/SDL3"
)

if(APPLE)
  list(APPEND TC_SDL3_CMAKE_ARGS
    "-DSDL_COCOA=ON"
    "-DSDL_METAL=ON"
  )
elseif(UNIX AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
  list(APPEND TC_SDL3_CMAKE_ARGS
    "-DSDL_X11=ON"
    "-DSDL_WAYLAND=ON"
    "-DSDL_KMSDRM=ON"
    "-DSDL_DBUS=ON"
    "-DSDL_IBUS=ON"
    "-DSDL_LIBUDEV=ON"
    "-DSDL_DEPS_SHARED=ON"
    "-DSDL_LIBURING=OFF"
    "-DSDL_FRIBIDI=OFF"
    "-DSDL_LIBTHAI=OFF"
    "-DSDL_RPI=OFF"
    "-DSDL_ROCKCHIP=OFF"
    "-DSDL_VIVANTE=OFF"
  )
endif()

set(TC_SDL3_EXTERNAL_PROJECT_OPTIONS)

if(DEFINED CMAKE_MSVC_RUNTIME_LIBRARY AND NOT CMAKE_MSVC_RUNTIME_LIBRARY STREQUAL "")
  list(APPEND TC_SDL3_CMAKE_ARGS
    "-DCMAKE_MSVC_RUNTIME_LIBRARY=${CMAKE_MSVC_RUNTIME_LIBRARY}")
endif()

if(CMAKE_GENERATOR_PLATFORM OR CMAKE_GENERATOR_TOOLSET)
  list(APPEND TC_SDL3_EXTERNAL_PROJECT_OPTIONS
    CMAKE_GENERATOR "${CMAKE_GENERATOR}"
  )
endif()
if(CMAKE_GENERATOR_PLATFORM)
  list(APPEND TC_SDL3_EXTERNAL_PROJECT_OPTIONS
    CMAKE_GENERATOR_PLATFORM "${CMAKE_GENERATOR_PLATFORM}"
  )
endif()
if(CMAKE_GENERATOR_TOOLSET)
  list(APPEND TC_SDL3_EXTERNAL_PROJECT_OPTIONS
    CMAKE_GENERATOR_TOOLSET "${CMAKE_GENERATOR_TOOLSET}"
  )
endif()

if(DEFINED CMAKE_TOOLCHAIN_FILE)
  list(APPEND TC_SDL3_CMAKE_ARGS "-DCMAKE_TOOLCHAIN_FILE=${CMAKE_TOOLCHAIN_FILE}")
endif()
if(CMAKE_OSX_SYSROOT)
  list(APPEND TC_SDL3_CMAKE_ARGS "-DCMAKE_OSX_SYSROOT=${CMAKE_OSX_SYSROOT}")
endif()
if(CMAKE_OSX_ARCHITECTURES)
  list(APPEND TC_SDL3_CMAKE_ARGS "-DCMAKE_OSX_ARCHITECTURES=${CMAKE_OSX_ARCHITECTURES}")
endif()
if(CMAKE_OSX_DEPLOYMENT_TARGET)
  list(APPEND TC_SDL3_CMAKE_ARGS "-DCMAKE_OSX_DEPLOYMENT_TARGET=${CMAKE_OSX_DEPLOYMENT_TARGET}")
endif()
if(CMAKE_SYSTEM_PROCESSOR)
  list(APPEND TC_SDL3_CMAKE_ARGS "-DCMAKE_SYSTEM_PROCESSOR=${CMAKE_SYSTEM_PROCESSOR}")
elseif(CMAKE_OSX_ARCHITECTURES)
  list(GET CMAKE_OSX_ARCHITECTURES 0 TC_SDL3_SYSTEM_PROCESSOR)
  list(APPEND TC_SDL3_CMAKE_ARGS "-DCMAKE_SYSTEM_PROCESSOR=${TC_SDL3_SYSTEM_PROCESSOR}")
endif()
if(CMAKE_SYSTEM_NAME)
  list(APPEND TC_SDL3_CMAKE_ARGS "-DCMAKE_SYSTEM_NAME=${CMAKE_SYSTEM_NAME}")
endif()

if(TC_SDL3_SOURCE_DIR)
  get_filename_component(TC_SDL3_SOURCE_DIR_ABSOLUTE "${TC_SDL3_SOURCE_DIR}" ABSOLUTE)
  if(NOT EXISTS "${TC_SDL3_SOURCE_DIR_ABSOLUTE}/include/SDL3/SDL.h" OR
     NOT EXISTS "${TC_SDL3_SOURCE_DIR_ABSOLUTE}/LICENSE.txt")
    message(FATAL_ERROR
      "TC_SDL3_SOURCE_DIR must contain include/SDL3/SDL.h and LICENSE.txt")
  endif()
  ExternalProject_Add(sdl3_external
    ${TC_SDL3_EXTERNAL_PROJECT_OPTIONS}
    SOURCE_DIR "${TC_SDL3_SOURCE_DIR_ABSOLUTE}"
    BINARY_DIR "${CMAKE_BINARY_DIR}/sdl3-build"
    INSTALL_DIR "${TC_SDL3_EXTERNAL_INSTALL_DIR}"
    CMAKE_ARGS ${TC_SDL3_CMAKE_ARGS}
  )
else()
  ExternalProject_Add(sdl3_external
    ${TC_SDL3_EXTERNAL_PROJECT_OPTIONS}
    URL "${TC_SDL3_SOURCE_URL}"
    URL_HASH "SHA256=${TC_SDL3_SOURCE_SHA256}"
    DOWNLOAD_EXTRACT_TIMESTAMP FALSE
    BINARY_DIR "${CMAKE_BINARY_DIR}/sdl3-build"
    INSTALL_DIR "${TC_SDL3_EXTERNAL_INSTALL_DIR}"
    CMAKE_ARGS ${TC_SDL3_CMAKE_ARGS}
  )
endif()

install(DIRECTORY "${TC_SDL3_EXTERNAL_INSTALL_DIR}/include/"
  DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
)
install(DIRECTORY "${TC_SDL3_EXTERNAL_INSTALL_DIR}/lib/"
  DESTINATION "${CMAKE_INSTALL_LIBDIR}"
)
if(WIN32 AND NOT MINGW)
  set(TC_SDL3_LICENSE_FILE
    "${TC_SDL3_EXTERNAL_INSTALL_DIR}/licenses/SDL3/LICENSE.txt")
else()
  set(TC_SDL3_LICENSE_FILE
    "${TC_SDL3_EXTERNAL_INSTALL_DIR}/share/licenses/SDL3/LICENSE.txt")
endif()
install(FILES "${TC_SDL3_LICENSE_FILE}"
  DESTINATION "${CMAKE_INSTALL_DATADIR}/licenses/sdl3")
