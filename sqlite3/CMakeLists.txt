cmake_minimum_required(VERSION 3.10)

project(totalcross-sqlite3-static
  VERSION 3.32.3
  LANGUAGES C
)

include(GNUInstallDirs)

set(SQLITE3_SOURCE_DIR "" CACHE PATH "Directory containing sqlite3.c and sqlite3.h")
set(SQLITE3_SOURCE_ARCHIVE "" CACHE FILEPATH "SQLite3 source archive containing sqlite3.c and sqlite3.h")
set(SQLITE3_SOURCE_URL "" CACHE STRING "SQLite3 source archive URL")

set(SQLITE3_YEAR 2020)
set(SQLITE3_VERSION 3.32.03.00)
string(REPLACE "." "" SQLITE3_FILE_VERSION "${SQLITE3_VERSION}")

if(SQLITE3_SOURCE_DIR)
  get_filename_component(sqlite3_SOURCE_DIR "${SQLITE3_SOURCE_DIR}" ABSOLUTE)
else()
  if(SQLITE3_SOURCE_ARCHIVE)
    get_filename_component(TC_SQLITE3_SOURCE_ARCHIVE "${SQLITE3_SOURCE_ARCHIVE}" ABSOLUTE)
  elseif(SQLITE3_SOURCE_URL)
    set(TC_SQLITE3_SOURCE_ARCHIVE "${CMAKE_BINARY_DIR}/sources/sqlite3-source")
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/sources")
    file(DOWNLOAD
      "${SQLITE3_SOURCE_URL}"
      "${TC_SQLITE3_SOURCE_ARCHIVE}"
      SHOW_PROGRESS
      STATUS TC_SQLITE3_DOWNLOAD_STATUS
    )
    list(GET TC_SQLITE3_DOWNLOAD_STATUS 0 TC_SQLITE3_DOWNLOAD_CODE)
    list(GET TC_SQLITE3_DOWNLOAD_STATUS 1 TC_SQLITE3_DOWNLOAD_MESSAGE)
    if(NOT TC_SQLITE3_DOWNLOAD_CODE EQUAL 0)
      message(FATAL_ERROR "Failed to download SQLite3 source: ${TC_SQLITE3_DOWNLOAD_MESSAGE}")
    endif()
  else()
    set(TC_SQLITE3_SOURCE_ARCHIVE
      "${CMAKE_BINARY_DIR}/sources/sqlite-amalgamation-${SQLITE3_FILE_VERSION}.zip"
    )
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/sources")
    file(DOWNLOAD
      "https://sqlite.org/${SQLITE3_YEAR}/sqlite-amalgamation-${SQLITE3_FILE_VERSION}.zip"
      "${TC_SQLITE3_SOURCE_ARCHIVE}"
      SHOW_PROGRESS
      STATUS TC_SQLITE3_DOWNLOAD_STATUS
    )
    list(GET TC_SQLITE3_DOWNLOAD_STATUS 0 TC_SQLITE3_DOWNLOAD_CODE)
    list(GET TC_SQLITE3_DOWNLOAD_STATUS 1 TC_SQLITE3_DOWNLOAD_MESSAGE)
    if(NOT TC_SQLITE3_DOWNLOAD_CODE EQUAL 0)
      message(FATAL_ERROR "Failed to download SQLite3 source: ${TC_SQLITE3_DOWNLOAD_MESSAGE}")
    endif()
  endif()

  set(sqlite3_SOURCE_DIR "${CMAKE_BINARY_DIR}/sqlite3-src")
  file(REMOVE_RECURSE "${sqlite3_SOURCE_DIR}")
  file(MAKE_DIRECTORY "${sqlite3_SOURCE_DIR}")
  execute_process(
    COMMAND "${CMAKE_COMMAND}" -E tar xzf "${TC_SQLITE3_SOURCE_ARCHIVE}"
    WORKING_DIRECTORY "${sqlite3_SOURCE_DIR}"
    RESULT_VARIABLE TC_SQLITE3_EXTRACT_RESULT
  )
  if(NOT TC_SQLITE3_EXTRACT_RESULT EQUAL 0)
    message(FATAL_ERROR "Failed to extract SQLite3 source archive: ${TC_SQLITE3_SOURCE_ARCHIVE}")
  endif()
endif()

file(GLOB_RECURSE TC_SQLITE3_CANDIDATE_SOURCES
  "${sqlite3_SOURCE_DIR}/sqlite3.c"
  "${sqlite3_SOURCE_DIR}/*/sqlite3.c"
)

file(GLOB_RECURSE TC_SQLITE3_CANDIDATE_HEADERS
  "${sqlite3_SOURCE_DIR}/sqlite3.h"
  "${sqlite3_SOURCE_DIR}/*/sqlite3.h"
)

list(REMOVE_DUPLICATES TC_SQLITE3_CANDIDATE_SOURCES)
list(REMOVE_DUPLICATES TC_SQLITE3_CANDIDATE_HEADERS)

list(LENGTH TC_SQLITE3_CANDIDATE_SOURCES TC_SQLITE3_SOURCE_COUNT)
list(LENGTH TC_SQLITE3_CANDIDATE_HEADERS TC_SQLITE3_HEADER_COUNT)

if(NOT TC_SQLITE3_SOURCE_COUNT EQUAL 1)
  message(FATAL_ERROR "Expected exactly one sqlite3.c under '${sqlite3_SOURCE_DIR}', found ${TC_SQLITE3_SOURCE_COUNT}: ${TC_SQLITE3_CANDIDATE_SOURCES}")
endif()

if(NOT TC_SQLITE3_HEADER_COUNT EQUAL 1)
  message(FATAL_ERROR "Expected exactly one sqlite3.h under '${sqlite3_SOURCE_DIR}', found ${TC_SQLITE3_HEADER_COUNT}: ${TC_SQLITE3_CANDIDATE_HEADERS}")
endif()

list(GET TC_SQLITE3_CANDIDATE_SOURCES 0 TC_SQLITE3_SOURCE_FILE)
list(GET TC_SQLITE3_CANDIDATE_HEADERS 0 TC_SQLITE3_HEADER_FILE)
get_filename_component(TC_SQLITE3_INCLUDE_DIR "${TC_SQLITE3_HEADER_FILE}" DIRECTORY)

add_library(SQLite3 STATIC "${TC_SQLITE3_SOURCE_FILE}")
add_library(SQLite3::SQLite3 ALIAS SQLite3)

target_include_directories(SQLite3
  PUBLIC
    "$<BUILD_INTERFACE:${TC_SQLITE3_INCLUDE_DIR}>"
    "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>"
)

target_compile_definitions(SQLite3
  PRIVATE
    SQLITE_ENABLE_COLUMN_METADATA=1
    SQLITE_ENABLE_FTS3=1
    SQLITE_ENABLE_LOAD_EXTENSION=1
    SQLITE_ENABLE_UPDATE_DELETE_LIMIT=1
    SQLITE_ENABLE_FTS3_PARENTHESIS=1
    SQLITE_ENABLE_RTREE=1
    SQLITE_ENABLE_STAT2=1
    SQLITE_ENABLE_UNLOCK_NOTIFY=1
    SQLITE_THREADSAFE=0
    SQLITE_THREAD_OVERRIDE_LOCK=0
    SQLITE_WITHOUT_MSIZE=1
    SQLITE_DQS=0
    SQLITE_DEFAULT_MEMSTATUS=0
    SQLITE_DEFAULT_WAL_SYNCHRONOUS=1
    QLITE_LIKE_DOESNT_MATCH_BLOBS
    SQLITE_MAX_EXPR_DEPTH=0
    SQLITE_OMIT_DEPRECATED
    SQLITE_OMIT_PROGRESS_CALLBACK
)

set_target_properties(SQLite3
  PROPERTIES
    OUTPUT_NAME sqlite3
    POSITION_INDEPENDENT_CODE ON
    PUBLIC_HEADER "${TC_SQLITE3_HEADER_FILE}"
    XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET 13.0
)

install(TARGETS SQLite3
  EXPORT SQLite3Targets
  ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}"
  PUBLIC_HEADER DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
)

install(FILES
  "${TC_SQLITE3_INCLUDE_DIR}/sqlite3ext.h"
  DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
  OPTIONAL
)

install(EXPORT SQLite3Targets
  NAMESPACE SQLite3::
  DESTINATION "${CMAKE_INSTALL_LIBDIR}/cmake/SQLite3"
)
