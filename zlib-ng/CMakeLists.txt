cmake_minimum_required(VERSION 3.16)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/TotalCrossWindowsStaticRuntime.cmake")

project(totalcross-zlib-ng-static
  VERSION 2.1.6
  LANGUAGES C
)

include(FetchContent)
include(GNUInstallDirs)

set(ZLIB_NG_SOURCE_DIR "" CACHE PATH "Directory containing zlib-ng sources")
set(ZLIB_NG_GIT_REPOSITORY "https://github.com/zlib-ng/zlib-ng.git" CACHE STRING "zlib-ng Git repository")
set(ZLIB_NG_GIT_TAG "2.1.6" CACHE STRING "zlib-ng Git tag")

set(ZLIB_COMPAT ON CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)

set(WITH_OPTIM ON CACHE BOOL "" FORCE)
set(WITH_REDUCED_MEM OFF CACHE BOOL "" FORCE)
set(WITH_NEW_STRATEGIES ON CACHE BOOL "" FORCE)
set(WITH_UNALIGNED ON CACHE BOOL "" FORCE)

if(DEFINED ANDROID_ABI)
  if(ANDROID_ABI STREQUAL "arm64-v8a")
    set(WITH_ARMV6 OFF CACHE BOOL "" FORCE)
  elseif(ANDROID_ABI STREQUAL "armeabi-v7a")
    set(WITH_NEON OFF CACHE BOOL "" FORCE)
  endif()
endif()

set(ZLIB_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(ZLIBNG_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_GTEST OFF CACHE BOOL "" FORCE)
set(WITH_FUZZERS OFF CACHE BOOL "" FORCE)
set(WITH_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(WITH_BENCHMARK_APPS OFF CACHE BOOL "" FORCE)
set(WITH_MAINTAINER_WARNINGS OFF CACHE BOOL "" FORCE)
set(WITH_CODE_COVERAGE OFF CACHE BOOL "" FORCE)
set(WITH_INFLATE_STRICT OFF CACHE BOOL "" FORCE)
set(WITH_INFLATE_ALLOW_INVALID_DIST OFF CACHE BOOL "" FORCE)

if(ZLIB_NG_SOURCE_DIR)
  set(zlib_SOURCE_DIR "${ZLIB_NG_SOURCE_DIR}")
else()
  FetchContent_Declare(
    zlib
    GIT_REPOSITORY "${ZLIB_NG_GIT_REPOSITORY}"
    GIT_TAG "${ZLIB_NG_GIT_TAG}"
    GIT_SHALLOW TRUE
    GIT_PROGRESS TRUE
    USES_TERMINAL_DOWNLOAD TRUE
  )

  FetchContent_GetProperties(zlib)
  if(NOT zlib_POPULATED)
    FetchContent_Populate(zlib)
  endif()
endif()

add_subdirectory("${zlib_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/zlib-ng-build")

get_target_property(ZLIBSTATIC_ALIASED_TARGET zlibstatic ALIASED_TARGET)
if(ZLIBSTATIC_ALIASED_TARGET)
  set(ZLIBSTATIC_TARGET "${ZLIBSTATIC_ALIASED_TARGET}")
else()
  set(ZLIBSTATIC_TARGET zlibstatic)
endif()

target_include_directories("${ZLIBSTATIC_TARGET}"
  PUBLIC
    "$<BUILD_INTERFACE:${CMAKE_BINARY_DIR}/zlib-ng-build>"
    "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>"
)

set_target_properties("${ZLIBSTATIC_TARGET}"
  PROPERTIES
    POSITION_INDEPENDENT_CODE ON
    XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET 13.0
)

install(TARGETS "${ZLIBSTATIC_TARGET}"
  ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}"
  LIBRARY DESTINATION "${CMAKE_INSTALL_LIBDIR}"
  RUNTIME DESTINATION "${CMAKE_INSTALL_BINDIR}"
)

install(FILES
  "${CMAKE_BINARY_DIR}/zlib-ng-build/zlib.h"
  "${CMAKE_BINARY_DIR}/zlib-ng-build/zconf.h"
  "${CMAKE_BINARY_DIR}/zlib-ng-build/zlib_name_mangling.h"
  DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}"
)
