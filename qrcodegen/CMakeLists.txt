cmake_minimum_required(VERSION 3.16)

project(totalcross-qrcodegen-static VERSION 20250123 LANGUAGES C)

include(GNUInstallDirs)
include(CTest)

set(QRCODEGEN_UPSTREAM_COMMIT "2c9044de6b049ca25cb3cd1649ed7e27aa055138")
set(QRCODEGEN_SOURCE_DIR "" CACHE PATH "Directory containing Nayuki QR-Code-generator sources")
set(QRCODEGEN_SOURCE_ARCHIVE "" CACHE FILEPATH "Nayuki QR-Code-generator source archive")
set(QRCODEGEN_SOURCE_URL
  "https://codeload.github.com/nayuki/QR-Code-generator/tar.gz/${QRCODEGEN_UPSTREAM_COMMIT}"
  CACHE STRING "Nayuki QR-Code-generator source archive URL")
set(QRCODEGEN_SOURCE_SHA256
  "fd5b5079196cbe4d733838e784eecc1d854f3772562f166483a9b617844922f5"
  CACHE STRING "Expected SHA-256 for the Nayuki source archive")

if(QRCODEGEN_SOURCE_DIR)
  get_filename_component(TC_QRCODEGEN_INPUT_SOURCE_DIR "${QRCODEGEN_SOURCE_DIR}" ABSOLUTE)
  if(NOT EXISTS "${TC_QRCODEGEN_INPUT_SOURCE_DIR}/c/qrcodegen.c")
    message(FATAL_ERROR "QRCODEGEN_SOURCE_DIR must contain c/qrcodegen.c")
  endif()
  set(TC_QRCODEGEN_UNPACK_DIR "${CMAKE_BINARY_DIR}/qrcodegen-unpacked")
  file(REMOVE_RECURSE "${TC_QRCODEGEN_UNPACK_DIR}")
  file(MAKE_DIRECTORY "${TC_QRCODEGEN_UNPACK_DIR}")
  file(COPY "${TC_QRCODEGEN_INPUT_SOURCE_DIR}/" DESTINATION "${TC_QRCODEGEN_UNPACK_DIR}/source")
  set(TC_QRCODEGEN_SOURCE_DIR "${TC_QRCODEGEN_UNPACK_DIR}/source")
else()
  if(QRCODEGEN_SOURCE_ARCHIVE)
    get_filename_component(TC_QRCODEGEN_SOURCE_ARCHIVE "${QRCODEGEN_SOURCE_ARCHIVE}" ABSOLUTE)
  else()
    file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/sources")
    set(TC_QRCODEGEN_SOURCE_ARCHIVE "${CMAKE_BINARY_DIR}/sources/qrcodegen-${QRCODEGEN_UPSTREAM_COMMIT}.tar.gz")
    file(DOWNLOAD "${QRCODEGEN_SOURCE_URL}" "${TC_QRCODEGEN_SOURCE_ARCHIVE}"
      EXPECTED_HASH "SHA256=${QRCODEGEN_SOURCE_SHA256}"
      SHOW_PROGRESS STATUS TC_QRCODEGEN_DOWNLOAD_STATUS)
    list(GET TC_QRCODEGEN_DOWNLOAD_STATUS 0 TC_QRCODEGEN_DOWNLOAD_CODE)
    list(GET TC_QRCODEGEN_DOWNLOAD_STATUS 1 TC_QRCODEGEN_DOWNLOAD_MESSAGE)
    if(NOT TC_QRCODEGEN_DOWNLOAD_CODE EQUAL 0)
      message(FATAL_ERROR "Failed to download QR-Code-generator source: ${TC_QRCODEGEN_DOWNLOAD_MESSAGE}")
    endif()
  endif()
  file(SHA256 "${TC_QRCODEGEN_SOURCE_ARCHIVE}" TC_QRCODEGEN_ARCHIVE_SHA256)
  if(NOT TC_QRCODEGEN_ARCHIVE_SHA256 STREQUAL QRCODEGEN_SOURCE_SHA256)
    message(FATAL_ERROR "QR-Code-generator source SHA-256 mismatch: ${TC_QRCODEGEN_ARCHIVE_SHA256}")
  endif()
  set(TC_QRCODEGEN_UNPACK_DIR "${CMAKE_BINARY_DIR}/qrcodegen-unpacked")
  file(REMOVE_RECURSE "${TC_QRCODEGEN_UNPACK_DIR}")
  file(MAKE_DIRECTORY "${TC_QRCODEGEN_UNPACK_DIR}")
  execute_process(COMMAND "${CMAKE_COMMAND}" -E tar xzf "${TC_QRCODEGEN_SOURCE_ARCHIVE}"
    WORKING_DIRECTORY "${TC_QRCODEGEN_UNPACK_DIR}" RESULT_VARIABLE TC_QRCODEGEN_EXTRACT_RESULT)
  if(NOT TC_QRCODEGEN_EXTRACT_RESULT EQUAL 0)
    message(FATAL_ERROR "Failed to extract QR-Code-generator source: ${TC_QRCODEGEN_SOURCE_ARCHIVE}")
  endif()
  file(GLOB TC_QRCODEGEN_EXTRACTED_DIRS "${TC_QRCODEGEN_UNPACK_DIR}/QR-Code-generator-*")
  list(LENGTH TC_QRCODEGEN_EXTRACTED_DIRS TC_QRCODEGEN_EXTRACTED_DIR_COUNT)
  if(NOT TC_QRCODEGEN_EXTRACTED_DIR_COUNT EQUAL 1)
    message(FATAL_ERROR "Expected one QR-Code-generator source directory")
  endif()
  list(GET TC_QRCODEGEN_EXTRACTED_DIRS 0 TC_QRCODEGEN_SOURCE_DIR)
endif()

find_program(TC_QRCODEGEN_PATCH_EXECUTABLE patch REQUIRED)
execute_process(
  COMMAND "${TC_QRCODEGEN_PATCH_EXECUTABLE}" -p1 --batch --forward -l
    -i "${CMAKE_CURRENT_SOURCE_DIR}/patches/0001-totalcross-windows-wince-compat.patch"
  WORKING_DIRECTORY "${TC_QRCODEGEN_SOURCE_DIR}"
  RESULT_VARIABLE TC_QRCODEGEN_PATCH_RESULT OUTPUT_VARIABLE TC_QRCODEGEN_PATCH_OUTPUT
  ERROR_VARIABLE TC_QRCODEGEN_PATCH_ERROR)
if(NOT TC_QRCODEGEN_PATCH_RESULT EQUAL 0)
  message(FATAL_ERROR "Failed to apply qrcodegen compatibility patch.\n${TC_QRCODEGEN_PATCH_OUTPUT}\n${TC_QRCODEGEN_PATCH_ERROR}")
endif()

if(WIN32)
  configure_file("${CMAKE_CURRENT_SOURCE_DIR}/extensions/windows/stdbool.h"
    "${TC_QRCODEGEN_SOURCE_DIR}/c/stdbool.h" COPYONLY)
endif()

add_library(QRCodeGen STATIC "${TC_QRCODEGEN_SOURCE_DIR}/c/qrcodegen.c")
add_library(QRCodeGen::QRCodeGen ALIAS QRCodeGen)
set_target_properties(QRCodeGen PROPERTIES
  OUTPUT_NAME qrcodegen
  POSITION_INDEPENDENT_CODE ON
  C_STANDARD 99
  C_STANDARD_REQUIRED ON
  XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET 13.0)
target_include_directories(QRCodeGen PUBLIC
  "$<BUILD_INTERFACE:${TC_QRCODEGEN_SOURCE_DIR}/c>"
  "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
install(TARGETS QRCodeGen ARCHIVE DESTINATION "${CMAKE_INSTALL_LIBDIR}")
install(FILES "${TC_QRCODEGEN_SOURCE_DIR}/c/qrcodegen.h" DESTINATION "${CMAKE_INSTALL_INCLUDEDIR}")

if(BUILD_TESTING)
  add_executable(qrcodegen-smoke-test tests/qrcodegen_smoke_test.c)
  target_link_libraries(qrcodegen-smoke-test PRIVATE QRCodeGen)
  add_test(NAME qrcodegen-smoke-test COMMAND qrcodegen-smoke-test)
endif()
